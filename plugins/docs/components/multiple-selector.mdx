```ts
import { MultipleSelector } from "@dashin-dev/dashin"
//...
;<MultipleSelector
  columnDef={{ lookup: ["option1", "option2", "option3"] }}
  onFilterChanged={(_rowId, value) => console.log(value)}
/>
```

## Used in table column filtering

```ts
import { PluginColumns, MultipleSelector } from "@dashin-dev/dashin"
export default ({ t }: PluginColumns): Column<Type>[] => [
  //...
  {
    title: t("Status"),
    field: "status",
    filterComponent: props => <MultipleSelector {...props} />
  }
]
```

_Refer to [plugin-example-blog/post/columns.tsx](https://github.com/xbuilder/bunadmin/blob/master/plugins/bunadmin-plugin-example-blog/post/columns.tsx)_

## Props

```ts
type Props = {

  onFilterChanged?: (rowId: string, value: any) => void
  valueToLowerCase?: boolean
  valueReplaceSpaces?: boolean
}
```
